        // Rapordaki birebir veri seti
        const dataset = [
            { id: "tr", name: "Turkey", pop: 85200000, gdp: 10600, agri: 0.061, ind: 0.320, serv: 0.619 },
            { id: "dz", name: "Algeria", pop: 44900000, gdp: 3800, agri: 0.132, ind: 0.458, serv: 0.410 },
            { id: "de", name: "Germany", pop: 83200000, gdp: 52000, agri: 0.008, ind: 0.303, serv: 0.689 },
            { id: "cn", name: "China", pop: 1412000000, gdp: 12600, agri: 0.073, ind: 0.393, serv: 0.534 },
            { id: "us", name: "USA", pop: 331000000, gdp: 70000, agri: 0.010, ind: 0.191, serv: 0.799 },
            { id: "jp", name: "Japan", pop: 125000000, gdp: 40000, agri: 0.011, ind: 0.290, serv: 0.699 }
        ];

        // Rapordaki Kosinüs Benzerlik Matrisi Değerleri
        const matrixCountries = ["Turkey", "Algeria", "Germany", "China", "USA", "Japan"];
        const similarityMatrix = [
            [1.0, 0.85, -0.6, 0.2, -0.8, -0.56],
            [0.85, 1.0, -0.8, 0.4, -0.85, -0.81],
            [-0.6, -0.8, 1.0, -0.8, 0.8, 0.99],
            [0.2, 0.4, -0.8, 1.0, -0.68, -0.8],
            [-0.8, -0.85, 0.8, -0.68, 1.0, 0.79],
            [-0.56, -0.81, 0.99, -0.8, 0.79, 1.0]
        ];

        let radarChartInstance = null;

        document.addEventListener("DOMContentLoaded", () => {
            const selector = document.getElementById("countrySelector");
            
            // Seçim kutusunu doldurma
            dataset.forEach(c => {
                selector.innerHTML += `<option value="${c.id}">${c.name}</option>`;
            });

            // Kosinüs Benzerlik Tablosunu oluşturma ve renklendirme
            const simBody = document.getElementById("similarityGridBody");
            similarityMatrix.forEach((row, rIdx) => {
                let htmlRow = `<tr><td class="fw-bold bg-light text-start">${matrixCountries[rIdx]}</td>`;
                row.forEach(val => {
                    let cellBg = "";
                    if(val === 1.0) cellBg = "background-color: #cbd5e1; color: #0f172a;";
                    else if(val > 0.8) cellBg = "background-color: #dcfce7; color: #166534;";
                    else if(val > 0) cellBg = "background-color: #f0fdf4; color: #166534;";
                    else cellBg = "background-color: #fee2e2; color: #991b1b;";

                    htmlRow += `<td class="similarity-cell" style="${cellBg}">${val.toFixed(2)}</td>`;
                });
                htmlRow += `</tr>`;
                simBody.innerHTML += htmlRow;
            });

            // Değişim olayını dinleme
            selector.addEventListener("change", (e) => updateDashboard(e.target.value));
            
            // İlk açılış verisi (Türkiye)
            updateDashboard("tr");

            // TÜİK Çizgi Grafik Entegrasyonu
            initTuikChart();

            // PCA Biplot Çizimi (Rapordaki Grafik Koordinatlarının Matematiksel İzdüşümü)
            initPcaBiplot();
        });

        function updateDashboard(id) {
            const country = dataset.find(c => c.id === id);
            if(!country) return;

            document.getElementById("lblPop").innerText = country.pop.toLocaleString();
            document.getElementById("lblGdp").innerText = "$" + country.gdp.toLocaleString();
            
            let dominant = "Hizmet";
            if(country.ind > country.agri && country.ind > country.serv) dominant = "Sanayi";
            if(country.agri > country.ind && country.agri > country.serv) dominant = "Tarım";
            document.getElementById("lblSector").innerText = dominant;

            // Radar Grafik Güncellemesi
            const ctx = document.getElementById('sectorRadarChart').getContext('2d');
            if(radarChartInstance) radarChartInstance.destroy();

            radarChartInstance = new Chart(ctx, {
                type: 'radar',
                data: {
                    labels: ['Tarım (Agriculture)', 'Sanayi (Industry)', 'Hizmet (Service)'],
                    datasets: [{
                        label: `${country.name} Sektörel Dağılım Payı`,
                        data: [country.agri, country.ind, country.serv],
                        backgroundColor: 'rgba(59, 130, 246, 0.2)',
                        borderColor: '#3b82f6',
                        borderWidth: 2,
                        pointBackgroundColor: '#1d4ed8'
                    }]
                },
                options: {
                    responsive: true,
                    scales: { r: { min: 0, max: 1 } }
                }
            });
        }

        function initTuikChart() {
            const ctx = document.getElementById('tuikLineChart').getContext('2d');
            new Chart(ctx, {
                type: 'line',
                data: {
                    labels: ['Ocak', 'Şubat', 'Mart', 'Nisan'],
                    datasets: [
                        {
                            label: 'İhracat (USD)',
                            data: [20329140000, 21030890032, 21896870203, 25408276191],
                            borderColor: '#10b981',
                            tension: 0.2,
                            fill: false
                        },
                        {
                            label: 'İthalat (USD)',
                            data: [28696382057, 30078262336, 33119359794, 33908621290],
                            borderColor: '#ef4444',
                            tension: 0.2,
                            fill: false
                        }
                    ]
                },
                options: { responsive: true }
            });
        }

        function initPcaBiplot() {
            const ctx = document.getElementById('pcaBiplotChart').getContext('2d');
            new Chart(ctx, {
                type: 'scatter',
                data: {
                    datasets: [
                        {
                            label: 'Ülkeler',
                            data: [
                                {x: -0.9, y: -0.8, label: 'Turkey'},
                                {x: -2.1, y: -1.2, label: 'Algeria'},
                                {x: 1.4, y: -0.1, label: 'Germany'},
                                {x: -2.4, y: 1.8, label: 'China'},
                                {x: 2.9, y: 0.4, label: 'USA'},
                                {x: 1.2, y: -0.1, label: 'Japan'}
                            ],
                            backgroundColor: '#3b82f6',
                            pointRadius: 8,
                            pointHoverRadius: 10,
                            datalabels: { display: true }
                        }
                    ]
                },
                options: {
                    responsive: true,
                    plugins: {
                        tooltip: {
                            callbacks: {
                                label: function(ctx) { return ctx.raw.label + ` (PC1: ${ctx.raw.x}, PC2: ${ctx.raw.y})`; }
                            }
                        }
                    },
                    scales: {
                        x: { min: -3, max: 4, title: { display: true, text: 'PC1 (76.73%)' } },
                        y: { min: -2, max: 3, title: { display: true, text: 'PC2 (18.35%)' } }
                    }
                }
            });
        }
